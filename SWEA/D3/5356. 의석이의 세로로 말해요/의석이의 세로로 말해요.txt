T = int(input())

for testcase in range(1,T+1):
    arr = [[-1] * 15 for _ in range(15)]
    for i in range(5):
        temp = list(input())
        for j in range(len(temp)):
            arr[i][j] = temp[j]

    answer = []

    for i in range(15):
        for j in range(15):
            if arr[j][i] != -1:
                answer.append(arr[j][i])

    print(f"#{testcase} {''.join(answer)}")
