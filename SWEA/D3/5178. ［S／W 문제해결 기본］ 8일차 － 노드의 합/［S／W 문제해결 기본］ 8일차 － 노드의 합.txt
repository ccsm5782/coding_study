T = int(input())

for testcase in range(1, T+1):
    N, M, L = map(int, input().split())
    # N: 노드의 개수
    # M: 리프 노드의 개수
    # L: 값을 출력할 노드
    arr = [0] * (N + 1)
    for i in range(M):
        number, value = map(int,input().split())
        arr[number] = value

    for i in range(N-M, 0, -1):
        if N < 2 * i + 1:
            arr[i] = arr[2 * i]
        else:
            arr[i] = arr[2 * i] + arr[2 * i + 1]


    print(f"#{testcase} {arr[L]}")
