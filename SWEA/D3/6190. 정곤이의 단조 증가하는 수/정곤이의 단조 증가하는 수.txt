T = int(input())

for tc in range(1, T+1):
    N = int(input())
    arr = list(map(int, input().split()))
    answer = -1

    for i in range(N-1):
        for j in range(i+1, N):
            temp = []
            num = 0
            num = arr[i] * arr[j]
            num = str(num)
            for k in range(len(num)):
                temp.append(num[k])
            increase = 1

            # 단조 증가하지 않는 부분이 있다면, 그냥 넘어감
            for k in range(len(temp) - 1):
                if temp[k] > temp[k+1]:
                    increase = 0
                    break
            if increase == 0:
                continue

            # 단조 증가하는 숫자였다면,
            num = int(num)
            if num > answer:
                answer = num

    # for i in range(N-1):

    #             if str_num[_] > str_num[_+1]:
    #                 increase = 0
    #                 break
    #         if increase:
    #             if answer < num:
    #                 answer = num

    print(f"#{tc} {answer}")
