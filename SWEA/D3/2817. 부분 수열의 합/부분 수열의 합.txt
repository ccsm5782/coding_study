T = int(input())

for tc in range(1, T+1):
    n, k = map(int, input().split())
    arr = list(map(int, input().split()))
    cnt = 0

    # 비트시프팅 풀이
    for i in range(1 << n):
        temp = []
        for j in range(n):
            if i & (1 << j):
                temp.append(arr[n-j-1])
        if sum(temp) == k:
            cnt += 1

    print(f"#{tc} {cnt}")
