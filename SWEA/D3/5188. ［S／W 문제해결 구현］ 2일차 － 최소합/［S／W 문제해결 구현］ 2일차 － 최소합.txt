def dfs(r, c, current_sum):
    global min_sum

    # if current_sum >= min_sum:
    #     return # 이미 구한 합보다 현재 합이 크다면 탐색 필요 없음

    if r == N-1 and c == N-1: #오른쪽 아래 끝에 도달했을 때
        min_sum = min(min_sum, current_sum)
        return

    if r+1 < N: # 아래 방향 이동 가능
        dfs(r+1, c, current_sum + arr[r + 1][c])

    if c+1 < N: # 오른쪽 방향 이동 가능
        dfs(r, c+1, current_sum + arr[r][c + 1])



T = int(input())
for tc in range(1, T+1):
    N = int(input())
    arr = [list(map(int, input().split())) for _ in range(N)]

    min_sum = 100000

    dfs(0, 0, arr[0][0]) # 시작점 정의

    print(f"#{tc} {min_sum}")
