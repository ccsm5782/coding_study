def dfs(n):
    global answer
    global sum_kcal
    global answer
    global sum_score

    for i in range(n, N):

        if visit[i] == 0 and sum_kcal + kcal[i] <= L:
            visit[i] = 1
            sum_kcal += kcal[i]
            sum_score += score[i]

            if sum_score > answer:
                answer = sum_score
            dfs(i+1)

            visit[i] = 0
            sum_kcal -= kcal[i]
            sum_score -= score[i]



T = int(input())

for tc in range(1, T+1):

    N, L = map(int, input().split())
    score = []
    kcal = []
    visit = [0] * N
    answer = 0
    sum_kcal = 0
    sum_score = 0

    for _ in range(N):
        s, k = map(int, input().split())
        score.append(s)
        kcal.append(k)

    dfs(0)

    print(f"#{tc} {answer}")
